cmake_minimum_required(VERSION 3.20)
project(ReproCaptureTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(OpenCV_DIR "${REPO_ROOT}/third_party/build/opencv-4.11.0" CACHE PATH "Offline OpenCV")
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
add_executable(ReproCaptureTests ../../tests/ReproCaptureTests.cpp ../../src/Diagnostics/ReproCapture.cpp)
target_include_directories(ReproCaptureTests PRIVATE ../../src ../../packages/nlohmann ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ReproCaptureTests PRIVATE ${OpenCV_LIBS} cabinet.lib)
if(MSVC)
 target_compile_options(ReproCaptureTests PRIVATE /source-charset:utf-8 /execution-charset:utf-8)
endif()
enable_testing()
add_test(NAME ReproCaptureTests COMMAND ReproCaptureTests)
set_tests_properties(ReproCaptureTests PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OpenCV_DIR}/bin")
