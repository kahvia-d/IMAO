cmake_minimum_required(VERSION 3.24)
project(IMaoCloudCaptureProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(IMAO_SOURCE_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(IMAO_DEPENDENCY_ROOT "${IMAO_SOURCE_ROOT}" CACHE PATH "Existing local dependencies (no downloads)")
find_package(OpenCV REQUIRED PATHS "${IMAO_DEPENDENCY_ROOT}/third_party/build/opencv-4.11.0" NO_DEFAULT_PATH)
enable_testing()
add_executable(IMaoCloudSelectionTests SelectionTests.cpp)
target_compile_options(IMaoCloudSelectionTests PRIVATE /utf-8)
add_test(NAME IMaoCloudSelectionTests COMMAND IMaoCloudSelectionTests)
add_executable(IMaoGameCaptureTargetTests ../../tests/GameCaptureTargetTests.cpp)
target_include_directories(IMaoGameCaptureTargetTests PRIVATE "${IMAO_SOURCE_ROOT}/IMao-Core/src")
target_compile_definitions(IMaoGameCaptureTargetTests PRIVATE NOMINMAX)
target_link_libraries(IMaoGameCaptureTargetTests PRIVATE user32)
add_test(NAME IMaoGameCaptureTargetTests COMMAND IMaoGameCaptureTargetTests)
add_executable(IMaoBrowserSurfaceProbe SurfaceProbe.cpp)
target_compile_definitions(IMaoBrowserSurfaceProbe PRIVATE NOMINMAX)
target_link_libraries(IMaoBrowserSurfaceProbe PRIVATE user32 ole32 oleaut32 uiautomationcore)

# Reuse the existing standalone sampler, not a second implementation of WGC.
add_library(CloudProbeCapture STATIC ../CoordinateOffline/CaptureEngine.cpp)
target_compile_options(CloudProbeCapture PRIVATE /utf-8 /bigobj)
target_compile_definitions(CloudProbeCapture PUBLIC NOMINMAX)
target_include_directories(CloudProbeCapture PUBLIC ${OpenCV_INCLUDE_DIRS}
    "${IMAO_SOURCE_ROOT}/IMao-Core/src" "${IMAO_DEPENDENCY_ROOT}/IMao-Core/packages/nlohmann"
    "${CMAKE_CURRENT_SOURCE_DIR}/../CoordinateOffline")
target_link_libraries(CloudProbeCapture PUBLIC ${OpenCV_LIBS} bcrypt d3d11 dxgi windowsapp runtimeobject dwmapi user32 gdi32 ole32)
add_executable(IMaoCloudCaptureFixture ../CoordinateOffline/SampleCaptureTests.cpp)
target_compile_options(IMaoCloudCaptureFixture PRIVATE /utf-8 /bigobj)
target_link_libraries(IMaoCloudCaptureFixture PRIVATE CloudProbeCapture)
add_executable(IMaoCloudCaptureProbe main.cpp)
target_compile_options(IMaoCloudCaptureProbe PRIVATE /utf-8 /bigobj)
target_link_libraries(IMaoCloudCaptureProbe PRIVATE CloudProbeCapture)
foreach(target IMaoCloudCaptureFixture IMaoCloudCaptureProbe)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:opencv_world>" "$<TARGET_FILE_DIR:${target}>")
endforeach()
target_compile_options(IMaoGameCaptureTargetTests PRIVATE /utf-8)

target_sources(IMaoCloudCaptureProbe PRIVATE ../../src/Runtime/BrowserSurface.cpp)
target_link_libraries(IMaoCloudCaptureProbe PRIVATE uiautomationcore oleaut32)
target_sources(IMaoBrowserSurfaceProbe PRIVATE ../../src/Runtime/BrowserSurface.cpp)
target_include_directories(IMaoBrowserSurfaceProbe PRIVATE "${IMAO_SOURCE_ROOT}/IMao-Core/src")
target_compile_options(IMaoBrowserSurfaceProbe PRIVATE /utf-8)
target_link_libraries(IMaoBrowserSurfaceProbe PRIVATE runtimeobject windowsapp)

add_executable(IMaoBrowserPreviewTests ../../tests/BrowserPreviewTests.cpp)
target_include_directories(IMaoBrowserPreviewTests PRIVATE "${IMAO_SOURCE_ROOT}/IMao-Core/src" ${OpenCV_INCLUDE_DIRS})
target_compile_options(IMaoBrowserPreviewTests PRIVATE /utf-8)
target_link_libraries(IMaoBrowserPreviewTests PRIVATE ${OpenCV_LIBS})
add_test(NAME IMaoBrowserPreviewTests COMMAND IMaoBrowserPreviewTests)
